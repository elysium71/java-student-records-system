# CLASS: Details

## CONSTANTS
    MIN_YEAR    = 1
    MAX_YEAR    = 4
    MIN_CWA     = 0.0
    MAX_CWA     = 100.0
    MIN_CREDITS = 0
    MAX_CREDITS = 400

## FIELDS (all PRIVATE)
    course    : String        // e.g., "BCompSci"
    yearLevel : Integer       // MIN_YEAR..MAX_YEAR
    cwa       : Double        // MIN_CWA..MAX_CWA
    status    : String        // "FT" or "PT"
    credits   : Integer       // MIN_CREDITS..MAX_CREDITS

## CONSTRUCTORS

### DEFAULT CONSTRUCTOR
    IMPORT: none
    EXPORT: none
    ASSERTION: Creates a Details with safe default values.
    ALGORITHM:
        course    <- "Unknown"
        yearLevel <- MIN_YEAR
        cwa       <- 0.0
        status    <- "FT"
        credits   <- 0
    END

### CONSTRUCTOR WITH PARAMETERS
    IMPORT: pCourse (String), pYearLevel (Integer), pCwa (Double),
            pStatus (String), pCredits (Integer)
    EXPORT: none
    ASSERTION: Creates Details with validated values.
    ALGORITHM:
        CALL setCourse(pCourse)
        CALL setYearLevel(pYearLevel)
        CALL setCwa(pCwa)
        CALL setStatus(pStatus)
        CALL setCredits(pCredits)
    END

### COPY CONSTRUCTOR
    IMPORT: pOther (Details)
    EXPORT: none
    ASSERTION: Deep-copies field values from pOther; if null, use defaults.
    ALGORITHM:
        IF pOther = NULL THEN
            course    <- "Unknown"
            yearLevel <- MIN_YEAR
            cwa       <- 0.0
            status    <- "FT"
            credits   <- 0
        ELSE
            course    <- pOther.getCourse()
            yearLevel <- pOther.getYearLevel()
            cwa       <- pOther.getCwa()
            status    <- pOther.getStatus()
            credits   <- pOther.getCredits()
        ENDIF
    END

## OBSERVER METHODS (GETTERS)

### METHOD: getCourse
    IMPORT: none
    EXPORT: String
    ASSERTION: Returns current course.
    ALGORITHM:
        RETURN course
    END

### METHOD: getYearLevel
    IMPORT: none
    EXPORT: Integer
    ASSERTION: Returns current year level.
    ALGORITHM:
        RETURN yearLevel
    END

### METHOD: getCwa
    IMPORT: none
    EXPORT: Double
    ASSERTION: Returns current CWA.
    ALGORITHM:
        RETURN cwa
    END

### METHOD: getStatus
    IMPORT: none
    EXPORT: String
    ASSERTION: Returns current enrolment status.
    ALGORITHM:
        RETURN status
    END

### METHOD: getCredits
    IMPORT: none
    EXPORT: Integer
    ASSERTION: Returns current credit total.
    ALGORITHM:
        RETURN credits
    END

## MUTATOR METHODS (SETTERS) - each validates input

### METHOD: setCourse
    IMPORT: pCourse (String)
    EXPORT: none
    ASSERTION: Course must be non-empty after trim.
    ALGORITHM:
        IF (pCourse = NULL) OR (trim(pCourse) = "") THEN
            ERROR "Course must be non-empty."
        ELSE
            course <- trim(pCourse)
        ENDIF
    END

### METHOD: setYearLevel
    IMPORT: pYearLevel (Integer)
    EXPORT: none
    ASSERTION: Year level within [MIN_YEAR..MAX_YEAR].
    ALGORITHM:
        IF (pYearLevel < MIN_YEAR) OR (pYearLevel > MAX_YEAR) THEN
            ERROR "Year level out of range."
        ELSE
            yearLevel <- pYearLevel
        ENDIF
    END

### METHOD: setCwa
    IMPORT: pCwa (Double)
    EXPORT: none
    ASSERTION: CWA within [MIN_CWA..MAX_CWA].
    ALGORITHM:
        IF (pCwa < MIN_CWA) OR (pCwa > MAX_CWA) THEN
            ERROR "CWA out of range."
        ELSE
            cwa <- pCwa
        ENDIF
    END

### METHOD: setStatus
    IMPORT: pStatus (String)
    EXPORT: none
    ASSERTION: Status is "FT" or "PT" (case-insensitive input).
    ALGORITHM:
        IF pStatus = NULL THEN
            ERROR "Status cannot be null."
        ELSE
            t <- UPPER(TRIM(pStatus))
            IF (t != "FT") AND (t != "PT") THEN
                ERROR "Status must be FT or PT."
            ELSE
                status <- t
            ENDIF
        ENDIF
    END

### METHOD: setCredits
    IMPORT: pCredits (Integer)
    EXPORT: none
    ASSERTION: Credits within [MIN_CREDITS..MAX_CREDITS].
    ALGORITHM:
        IF (pCredits < MIN_CREDITS) OR (pCredits > MAX_CREDITS) THEN
            ERROR "Credits out of range."
        ELSE
            credits <- pCredits
        ENDIF
    END



# END CLASS: Details




# CLASS: Student

## FIELDS (all PRIVATE)
    studentID : String   // unique student identifier
    firstName : String   // student's first name
    lastName  : String   // student's last name
    info      : Details  // has-a relationship: each student owns a Details object

## CONSTRUCTORS

### DEFAULT CONSTRUCTOR
    IMPORT: none
    EXPORT: none
    ASSERTION: Creates a Student with safe placeholder values.
    ALGORITHM:
        studentID <- "UNKNOWN"
        firstName <- "Unknown"
        lastName  <- "Unknown"
        info <- new Details()  // default Details instance
    END

### CONSTRUCTOR WITH PARAMETERS
    IMPORT: pID (String), pFirst (String), pLast (String), pInfo (Details)
    EXPORT: none
    ASSERTION: Creates a Student with validated identity fields and Details copy.
    ALGORITHM:
        CALL setStudentID(pID)
        CALL setFirstName(pFirst)
        CALL setLastName(pLast)
        IF pInfo = NULL THEN
            info <- new Details()
        ELSE
            info <- new Details(pInfo)   // deep copy for encapsulation
        ENDIF
    END


## OBSERVER METHODS (GETTERS)

### METHOD: getStudentID
    IMPORT: none
    EXPORT: String
    ASSERTION: Returns student ID.
    ALGORITHM:
        RETURN studentID
    END

### METHOD: getFirstName
    IMPORT: none
    EXPORT: String
    ASSERTION: Returns first name.
    ALGORITHM:
        RETURN firstName
    END

### METHOD: getLastName
    IMPORT: none
    EXPORT: String
    ASSERTION: Returns last name.
    ALGORITHM:
        RETURN lastName
    END

### METHOD: getInfo
    IMPORT: none
    EXPORT: Details
    ASSERTION: Returns a deep copy of Details for safety.
    ALGORITHM:
        RETURN new Details(info)
    END

## MUTATOR METHODS (SETTERS)

### METHOD: setStudentID
    IMPORT: pID (String)
    EXPORT: none
    ASSERTION: ID must be non-empty string.
    ALGORITHM:
        IF pID = NULL OR TRIM(pID) = "" THEN
            ERROR "StudentID must be non-empty."
        ELSE
            studentID <- TRIM(pID)
        ENDIF
    END

### METHOD: setFirstName
    IMPORT: pFirst (String)
    EXPORT: none
    ASSERTION: First name must be non-empty string.
    ALGORITHM:
        IF pFirst = NULL OR TRIM(pFirst) = "" THEN
            ERROR "First name must be non-empty."
        ELSE
            firstName <- TRIM(pFirst)
        ENDIF
    END

### METHOD: setLastName
    IMPORT: pLast (String)
    EXPORT: none
    ASSERTION: Last name must be non-empty string.
    ALGORITHM:
        IF pLast = NULL OR TRIM(pLast) = "" THEN
            ERROR "Last name must be non-empty."
        ELSE
            lastName <- TRIM(pLast)
        ENDIF
    END

### METHOD: setInfo
    IMPORT: pInfo (Details)
    EXPORT: none
    ASSERTION: Info object cannot be null; copy stored.
    ALGORITHM:
        IF pInfo = NULL THEN
            ERROR "Details cannot be null."
        ELSE
            info <- new Details(pInfo)
        ENDIF
    END

## UTILITY METHODS

### METHOD: getFullName
    IMPORT: none
    EXPORT: String
    ASSERTION: Combines first and last names with space.
    ALGORITHM:
        RETURN firstName + " " + lastName
    END



# END CLASS: Student

# Main Driver & Menu

## MAIN

    IMPORT: none
    EXPORT: none
    ALGORITHM:
        DECLARE students : ARRAY OF Student SIZE MAX_STUDENTS
        DECLARE usedCount : INTEGER <- 0
        usedCount <- loadFromCSV(DATA_FILENAME, students)

        DO
            userChoice <- menuAndGetChoice()
            CASE userChoice OF
                1: addStudent(students, usedCount)
                2: editStudent(students, usedCount)
                3: viewAll(students, usedCount)
                4: filterByCourse(students, usedCount)
                5: filterByStatus(students, usedCount)
                6: highestCWA(students, usedCount)
                7: averageCWAByCourse(students, usedCount)
                8: creditAnalysis(students, usedCount)
                9: saveToCSV(DATA_FILENAME, students, usedCount)
                   OUTPUT "Goodbye!"
            ENDCASE
        WHILE userChoice != 9
    ASSERTION: On exit, latest data saved to CSV.
    END MAIN

## METHOD: menuAndGetChoice

    IMPORT: none
    EXPORT: choice (Integer)
    ASSERTION: 1 <= choice <= 9
    ALGORITHM:
        REPEAT
            OUTPUT "--- Student Records Menu ---"
            OUTPUT "1) Add student"
            OUTPUT "2) Edit student"
            OUTPUT "3) View all"
            OUTPUT "4) Filter by course"
            OUTPUT "5) Filter by status"
            OUTPUT "6) Highest CWA"
            OUTPUT "7) Average CWA by course"
            OUTPUT "8) Credit analysis"
            OUTPUT "9) Exit"
            choice <- readInt("Enter choice 1-9: ")
        UNTIL (choice >= 1) AND (choice <= 9)
    END

# File I/O (CSV)

## METHOD: loadFromCSV

    IMPORT: filename (String), students (ARRAY OF Student)
    EXPORT: count (Integer)
    ASSERTION: Return number of students loaded (return 0 if file missing/empty)
    ALGORITHM:
        count <- 0
        TRY
            OPEN filename FOR READ
            line <- readLine()
            WHILE line != End Of File AND count < MAX_STUDENTS
                IF isBlank(line) THEN
                    line <- readLine()
                ELSE
                    s <- parseStudent(line)
                    IF isValidStudent(s) THEN
                        students[count] <- s
                        count <- count + 1
                    ENDIF
                    line <- readLine()
                ENDIF
            ENDWHILE
            CLOSE file
        CATCH anyFileError
            OUTPUT "Warning: could not load file, starting empty."
        ENDTRY
        RETURN count
    END

## METHOD: saveToCSV

    IMPORT: filename (String), students (Array OF Student), count (Integer)
    EXPORT: none
    ASSERTION: Writes exactly 'count' rows in CSV format
    ALGORITHM:
        TRY
            OPEN filename FOR WRITE (overwrite)
            FOR i <- 0 TO count-1
                row <- formatCSVRow(students[i])
                writeLine(row)
            ENDFOR
            CLOSE file
            OUTPUT "Saved " + count + " records."
        CATCH anyFileError
            OUTPUT "Error: could not save file."
        ENDTRY
    END

## METHOD: parseStudent

    IMPORT: csvRow (String)
    EXPORT: s (Student)
    ASSERTION: Returns Student with parsed fields
    ALGORITHM:
        tokens[] <- split(csvRow, ",")
        s.studentID <- trim(tokens[0])
        s.firstName <- trim(tokens[1])
        s.lastName  <- trim(tokens[2])
        s.info.course    <- trim(tokens[3])
        s.info.yearLevel <- toIntOrDefault(tokens[4], 0)
        s.info.cwa       <- toDoubleOrDefault(tokens[5], -1.0)
        s.info.status    <- trim(tokens[6])
        s.info.credits   <- toIntOrDefault(tokens[7], -1)
        RETURN s
    END

## METHOD: formatCSVRow

    IMPORT: s (Student)
    EXPORT: row (String)
    ASSERTION: Produces a single valid CSV line
    ALGORITHM:
        row <- joinWithComma([s.studentID, s.firstName, s.lastName,
                               s.info.course, toString(s.info.yearLevel),
                               toFixed2(s.info.cwa), s.info.status,
                               toString(s.info.credits)])
        RETURN row
    END

# Main - Core Operations

## METHOD: addStudent

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Appends a new valid Student if capacity allows
    ALGORITHM:
        IF usedCount = MAX_STUDENTS THEN
            OUTPUT "Database full."
            RETURN
        ENDIF

        DECLARE s : Student

        REPEAT
            s.studentID <- readString("Student ID: ")
        UNTIL (NOT isBlank(s.studentID)) AND (findIndexByID(students, usedCount, s.studentID) = -1)

        s.firstName <- readNonEmptyString("First name: ")
        s.lastName  <- readNonEmptyString("Last name: ")

        s.info.course    <- readNonEmptyString("Course: ")
        s.info.yearLevel <- readIntInRange("Year level (1-4): ", 1, 4)
        s.info.cwa       <- readDoubleInRange("CWA (0-100): ", 0.0, 100.0)
        s.info.status    <- readStatus("Status "FT","PT"): "
        s.info.credits   <- readIntInRange("Credits (0-400): ", 0, 400)

        students[usedCount] <- s
        usedCount <- usedCount + 1
        OUTPUT "Added."
    END

## METHOD: editStudent

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Updates an existing student in place
    ALGORITHM:
        IF usedCount = 0 THEN
            OUTPUT "No records."
            RETURN
        ENDIF
        id <- readString("Enter Student ID to edit: ")
        idx <- findIndexByID(students, usedCount, id)
        IF idx = -1 THEN
            OUTPUT "Not found."
            RETURN
        ENDIF

        s <- students[idx]
        OUTPUT "Press Enter to keep current value."

        newFirst <- readOptionalString("First name ["+s.firstName+"]: ")
        IF NOT isBlank(newFirst) THEN s.firstName <- newFirst 
        ENDIF

        newLast  <- readOptionalString("Last name ["+s.lastName+"]: ")
        IF NOT isBlank(newLast) THEN s.lastName <- newLast 
        ENDIF

        newCourse <- readOptionalString("Course ["+s.info.course+"]: ")
        IF NOT isBlank(newCourse) THEN s.info.course <- newCourse 
        ENDIF

        newYear <- readOptionalIntInRange("Year ["+s.info.yearLevel+"] 1-4: ", 1, 4)
        IF newYear != BLANK THEN s.info.yearLevel <- newYear 
        ENDIF

        newCwa <- readOptionalDoubleInRange("CWA ["+s.info.cwa+"] 0-100: ", 0.0, 100.0)
        IF newCwa != BLANK THEN s.info.cwa <- newCwa 
        ENDIF

        newStatus <- readOptionalStatus("Status ["+s.info.status+"]: ")
        IF newStatus != BLANK THEN s.info.status <- newStatus 
        ENDIF

        newCredits <- readOptionalIntInRange("Credits ["+s.info.credits+"] 0-400: ", 0, 400)
        IF newCredits != BLANK THEN s.info.credits <- newCredits 
        ENDIF

        students[idx] <- s
        OUTPUT "Updated."
    END

## METHOD: viewAll

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Displays all students
    ALGORITHM:
        IF usedCount = 0 THEN
            OUTPUT "No records."
            RETURN
        ENDIF
        OUTPUT "ID, First, Last, Course, Year, CWA, Status, Credits"
        FOR i <- 0 TO usedCount-1
            printStudentLine(students[i])
        ENDFOR
    END

## METHOD: filterByCourse

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Shows only students in a given course
    ALGORITHM:
        target <- readNonEmptyString("Course to filter: ")
        found <- FALSE
        FOR i <- 0 TO usedCount-1
            IF equalsIgnoreCase(students[i].info.course, target) THEN
                printStudentLine(students[i])
                found <- TRUE
            ENDIF
        ENDFOR
        IF NOT found THEN OUTPUT "No matches." ENDIF
    END

## METHOD: filterByStatus

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Shows only students with matching status
    ALGORITHM:
        target <- readStatus("Status to filter: ")
        found <- FALSE
        FOR i <- 0 TO usedCount-1
            IF equalsIgnoreCase(students[i].info.status, target) THEN
                printStudentLine(students[i])
                found <- TRUE
            ENDIF
        ENDFOR
        IF NOT found THEN OUTPUT "No matches." ENDIF
    END

## METHOD: highestCWA

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Displays student(s) with maximum CWA
    ALGORITHM:
        IF usedCount = 0 THEN OUTPUT "No records." ; RETURN ENDIF
        maxCwa <- students[0].info.cwa
        FOR i <- 1 TO usedCount-1
            IF students[i].info.cwa > maxCwa THEN
                maxCwa <- students[i].info.cwa
            ENDIF
        ENDFOR
        OUTPUT "Highest CWA = " + toFixed2(maxCwa)
        FOR i <- 0 TO usedCount-1
            IF approxEqual(students[i].info.cwa, maxCwa, 0.0001) THEN
                printStudentLine(students[i])
            ENDIF
        ENDFOR
    END

## METHOD: averageCWAByCourse

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Shows average CWA for each course
    ALGORITHM:
        IF usedCount = 0 THEN OUTPUT "No records." ; RETURN ENDIF

        DECLARE courseNames[MAX_STUDENTS] : String
        DECLARE courseSum[MAX_STUDENTS]   : Double
        DECLARE courseCnt[MAX_STUDENTS]   : Integer
        uniqueCount <- 0

        FOR i <- 0 TO usedCount-1
            c <- students[i].info.course
            pos <- indexOf(courseNames, uniqueCount, c)
            IF pos = -1 THEN
                courseNames[uniqueCount] <- c
                courseSum[uniqueCount]   <- students[i].info.cwa
                courseCnt[uniqueCount]   <- 1
                uniqueCount <- uniqueCount + 1
            ELSE
                courseSum[pos] <- courseSum[pos] + students[i].info.cwa
                courseCnt[pos] <- courseCnt[pos] + 1
            ENDIF
        ENDFOR

        FOR j <- 0 TO uniqueCount-1
            avg <- courseSum[j] / courseCnt[j]
            OUTPUT courseNames[j] + ": " + toFixed2(avg) + " (" + courseCnt[j] + " students)"
        ENDFOR
    END

## METHOD: creditAnalysis

    IMPORT: students (Array OF Student), usedCount (Integer)
    EXPORT: none
    ASSERTION: Show if students eligible for graduation (>= 400 credits)
    ALGORITHM:
        IF usedCount = 0 THEN OUTPUT "No records." ; RETURN ENDIF
        gradCredits <- 400
        FOR i <- 0 TO usedCount-1
            eligible <- (students[i].info.credits >= gradCredits)
            OUTPUT students[i].studentID + " " + fullName(students[i]) +
                   " Credits: " + students[i].info.credits +
                   " Eligible: " + (IF eligible THEN "Y" ELSE "N")
        ENDFOR
    END

# Utility Methods

## METHOD: findIndexByID

    IMPORT: students (Array OF Student), usedCount (Integer), id (String)
    EXPORT: index (Integer)
    ASSERTION: Returns position or -1 if not found
    ALGORITHM:
        FOR i <- 0 TO usedCount-1
            IF equalsIgnoreCase(students[i].studentID, id) THEN
                RETURN i
            ENDIF
        ENDFOR
        RETURN -1
    END

## METHOD: equalsIgnoreCase

    ALGORITHM equalsIgnoreCase
    IMPORT: a (STRING or NIL), b (STRING or NIL)
    EXPORT: isEqual (BOOLEAN)
    ASSERTION: Returns FALSE if either is NIL; otherwise compares a and b case-insensitively.

    IF a = NIL OR b = NIL THEN
        RETURN FALSE
    ENDIF
    RETURN ( TO_UPPER(TRIM(a)) = TO_UPPER(TRIM(b)) )
    END ALGORITHM


## METHOD: indexOfString

    ALGORITHM indexOfString
    IMPORT: arr (ARRAY of STRING), used (INTEGER), key (STRING)
    EXPORT: idx (INTEGER)
    ASSERTION: Searches only the first 'used' elements; returns −1 if not found.

    FOR i <- 0 TO used - 1 DO
        IF equalsIgnoreCase(arr[i], key) = TRUE THEN
        RETURN i
        ENDIF
      ENDFOR
    RETURN -1
    END ALGORITHM


## METHOD: printHeader

    ALGORITHM printHeader
      IMPORT: -
      EXPORT: -
      ASSERTION: Prints a fixed header line matching printStudentLine order.

      OUTPUT "ID, First, Last, Course, Year, CWA, Status, Credits"
    END ALGORITHM


## METHOD: printStudentLine

    ALGORITHM printStudentLine
      IMPORT: s (Student)
      EXPORT: -
      ASSERTION: Prints one student record; CWA to 2 decimal places.

      d <- s.getInfo()
      OUTPUT s.getStudentID() + ", " +
             s.getFirstName() + ", " +
             s.getLastName()  + ", " +
             d.getCourse()    + ", " +
             d.getYearLevel() + ", " +
             FORMAT(d.getCwa(), 2dp) + ", " +
             d.getStatus()    + ", " +
             d.getCredits()
    END ALGORITHM


## METHOD: readLine

    ALGORITHM readLine
      IMPORT: sc (Scanner), prompt (STRING)
      EXPORT: line (STRING)
      ASSERTION: Returns the raw line (may be empty).

      OUTPUT_NO_NEWLINE prompt
      line <- SCAN_LINE(sc)
      RETURN line
    END ALGORITHM

## METHOD: readNonEmptyLine

    ALGORITHM readNonEmptyLine
      IMPORT: sc (Scanner), prompt (STRING)
      EXPORT: line (STRING)
      ASSERTION: Re-prompts until user enters a non-empty string.

      REPEAT
        OUTPUT_NO_NEWLINE prompt
        line <- TRIM( SCAN_LINE(sc) )
      UNTIL LENGTH(line) > 0
      RETURN line
    END ALGORITHM


## METHOD: readOptionalLine

    ALGORITHM readOptionalLine
      IMPORT: sc (Scanner), prompt (STRING)
      EXPORT: line (STRING)
      ASSERTION: Returns trimmed text; may be empty string.

      OUTPUT_NO_NEWLINE prompt
      line <- TRIM( SCAN_LINE(sc) )
      RETURN line
    END ALGORITHM


## METHOD: readInt  

    ALGORITHM readInt
      IMPORT: sc (Scanner), prompt (STRING)
      EXPORT: x (INTEGER)
      ASSERTION: Re-prompts until a valid whole number is entered.

      LOOP
        OUTPUT_NO_NEWLINE prompt
        t <- TRIM( SCAN_LINE(sc) )
        IF CAN_PARSE_INT(t) = TRUE THEN
          x <- PARSE_INT(t)
          RETURN x
        ELSE
          OUTPUT "Please enter a whole number."
        ENDIF
      ENDLOOP
    END ALGORITHM

## METHOD: readDouble  

ALGORITHM readDouble
      IMPORT: sc (Scanner), prompt (STRING)
      EXPORT: x (REAL)
      ASSERTION: Re-prompts until a valid real number is entered.

      LOOP
        OUTPUT_NO_NEWLINE prompt
        t <- TRIM( SCAN_LINE(sc) )
        IF CAN_PARSE_REAL(t) = TRUE THEN
          x <- PARSE_REAL(t)
          RETURN x
        ELSE
          OUTPUT "Please enter a valid number."
        ENDIF
      ENDLOOP
    END ALGORITHM

## METHOD: readIntInRange  

    ALGORITHM readIntInRange
      IMPORT: sc (Scanner), prompt (STRING), lo (INTEGER), hi (INTEGER)
      EXPORT: x (INTEGER)
      ASSERTION: Returns an integer x such that lo <= x <= hi.

      REPEAT
        x <- readInt(sc, prompt)
      UNTIL (x >= lo) AND (x <= hi)
      RETURN x
    END ALGORITHM

## METHOD: readDoubleInRange  

    ALGORITHM readDoubleInRange
      IMPORT: sc (Scanner), prompt (STRING), lo (REAL), hi (REAL)
      EXPORT: x (REAL)
      ASSERTION: Returns a real x such that lo <= x <= hi.

      REPEAT
        x <- readDouble(sc, prompt)
      UNTIL (x => lo) AND (x <= hi)
      RETURN x
    END ALGORITHM

## METHOD: readStatus  

    ALGORITHM readStatus
      IMPORT: sc (Scanner), prompt (STRING)
      EXPORT: status (STRING)
      ASSERTION: Accepts only "FT" or "PT" (case-insensitive); returns uppercase.

      REPEAT
        OUTPUT_NO_NEWLINE prompt
        status <- TO_UPPER( TRIM( SCAN_LINE(sc) ) )
      UNTIL (status = "FT") OR (status = "PT")
      RETURN status
    END ALGORITHM
